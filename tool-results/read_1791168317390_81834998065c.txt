   421→// -------- SOC Stock Tab --------
   422→interface LayerState {
   423→  socPct: string;
   424→  bulkDensity: string;
   425→  depthCm: string;
   426→  coarseFragPct: string;
   427→}
   428→
   429→function SocStockTab() {
   430→  const { t, fmt } = useI18n();
   431→  const [layers, setLayers] = useState<LayerState[]>([
   432→    { socPct: "1.85", bulkDensity: "1.35", depthCm: "30", coarseFragPct: "5" },
   433→  ]);
   434→  const [result, setResult] = useState<{ socStockTCPerHa: number; co2eTPerHa: number; byLayer: { depthCm: number; stockTCPerHa: number }[] } | null>(null);
   435→  const [loading, setLoading] = useState(false);
   436→
   437→  function addLayer() {
   438→    setLayers([...layers, { socPct: "", bulkDensity: "1.35", depthCm: "10", coarseFragPct: "0" }]);
   439→  }
   440→  function removeLayer(i: number) {
   441→    setLayers(layers.filter((_, idx) => idx !== i));
   442→  }
   443→  function update(i: number, key: keyof LayerState, v: string) {
   444→    setLayers(layers.map((l, idx) => (idx === i ? { ...l, [key]: v } : l)));
   445→  }
   446→
   447→  async function compute() {
   448→    setLoading(true);
   449→    try {
   450→      const body = {
   451→        layers: layers.map((l) => ({
   452→          socPct: Number(l.socPct),
   453→          bulkDensity: Number(l.bulkDensity),
   454→          depthCm: Number(l.depthCm),
   455→          coarseFragPct: Number(l.coarseFragPct) || 0,
   456→        })),
   457→      };
   458→      const res = await fetch("/api/calculate/soc-stock", {
   459→        method: "POST",
   460→        headers: { "Content-Type": "application/json" },
   461→        body: JSON.stringify(body),
   462→      });
   463→      if (!res.ok) throw new Error("Failed");
   464→      setResult(await res.json());

  1100→/**
  1101→ * Like Metric, but shows before/after delta when dMRV is on. If dMRV is off
  1102→ * (dmrvOn=false), behaves like a single-value Metric (no delta shown).
  1103→ */
  1104→function MetricCompare({
  1105→  label,
  1106→  baseline,
  1107→  adjusted,
  1108→  fmt,
  1109→  suffix,
  1110→  tone,
  1111→  dmrvOn,
  1112→  lowerIsBetter = false,
  1113→}: {
  1114→  label: string;
  1115→  baseline: number | null | undefined;
  1116→  adjusted: number | null;
  1117→  fmt: (n: number, opts?: Intl.NumberFormatOptions) => string;
  1118→  suffix: string;
  1119→  tone: "violet" | "emerald" | "amber" | "sky";
  1120→  dmrvOn: boolean;
  1121→  lowerIsBetter?: boolean;
  1122→}) {
  1123→  const tones: Record<string, string> = {
  1124→    violet: "bg-violet-50/60 dark:bg-violet-950/20",
  1125→    emerald: "bg-emerald-50/60 dark:bg-emerald-950/20",
  1126→    amber: "bg-amber-50/60 dark:bg-amber-950/20",
  1127→    sky: "bg-sky-50/60 dark:bg-sky-950/20",
  1128→  };
  1129→
  1130→  // Compute delta
  1131→  const hasBoth = dmrvOn && baseline != null && adjusted != null;
  1132→  const delta = hasBoth ? (adjusted as number) - (baseline as number) : 0;
  1133→  // "Improved" means delta is in the good direction (lower is better => negative delta good)
  1134→  const isImproved = hasBoth && (lowerIsBetter ? delta < 0 : delta > 0);
  1135→  const isFlat = hasBoth && delta === 0;
  1136→  const deltaLabel =
  1137→    delta > 0 ? `+${fmt(delta)}` : `${fmt(delta)}`;
  1138→
  1139→  return (
  1140→    <div className={`rounded-md p-3 ${tones[tone]}`}>
  1141→      <div className="text-xs text-muted-foreground">{label}</div>
  1142→      <div className="text-xl font-semibold tabular-nums">
  1143→        {adjusted === null ? "—" : fmt(adjusted)}
  1144→      </div>
  1145→      <div className="flex items-center gap-1 text-xs">
  1146→        {hasBoth ? (
  1147→          <>
  1148→            <span className="text-muted-foreground">
  1149→              {fmt(baseline as number)} →
  1150→            </span>
  1151→            <span
  1152→              className={
  1153→                isImproved
  1154→                  ? "text-emerald-600 dark:text-emerald-400 font-medium"
  1155→                  : isFlat
  1156→                  ? "text-muted-foreground"
  1157→                  : "text-red-600 dark:text-red-400 font-medium"
  1158→              }
  1159→            >
  1160→              {isImproved ? <ArrowUpRight className="inline h-3 w-3" /> : null}
  1161→              {isImproved || !isFlat ? (
  1162→                <ArrowDownRight className={isImproved ? "hidden" : "inline h-3 w-3"} />
  1163→              ) : null}
  1164→              {" "}
  1165→              {deltaLabel}
  1166→            </span>
  1167→          </>
  1168→        ) : (
  1169→          <span className="text-muted-foreground">{suffix}</span>
  1170→        )}
  1171→      </div>
  1172→    </div>
  1173→  );
  1174→}
